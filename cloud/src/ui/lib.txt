  { key: "video", label: "Video + text", implemented: false, note: "Not implemented (Videos API upload is not built)." },

  { key: "poll", label: "Poll", implemented: false, note: "Not implemented." },

const TYPE_BY_KEY = Object.fromEntries(CONTENT_TYPES.map((t) => [t.key, t]));
const FORMAT_TYPE = { video: "video", document: "document", pdf: "document", carousel: "document",
  article: "article", link: "article", multi_image: "multi_image", "multi-image": "multi_image", poll: "poll" };

/** Content type of a post, from its declared format and its image decision. */
export function contentType(post) {
  const fmt = String(post?.format ?? "").toLowerCase();
  if (FORMAT_TYPE[fmt]) return TYPE_BY_KEY[FORMAT_TYPE[fmt]];
  const img = post?.image;
  const hasImage = img && (img.kind ? img.kind !== "none" : Boolean(img.sha256 || img.bytes));
  return TYPE_BY_KEY[hasImage ? "image" : "text"];
}

/** First meaningful line, shortened: the hook shown in lists and the calendar. */
export function hook(text, max = 90) {
  const line = String(text ?? "").split("\n").map((l) => l.trim()).find(Boolean);
  if (!line) return null;
  return line.length > max ? line.slice(0, max - 1).trimEnd() + "…" : line;
}

export function hashtags(text) {
  return [...new Set(String(text ?? "").match(/(^|\s)#[\p{L}\p{N}_]+/gu)?.map((t) => t.trim()) ?? [])];
}

/** Split text into plain and hashtag segments for the feed preview. */
export function segments(text) {
  const out = [];
  const re = /#[\p{L}\p{N}_]+/gu;
  let last = 0;
  const s = String(text ?? "");
  for (const m of s.matchAll(re)) {
    if (m.index > last) out.push({ text: s.slice(last, m.index), tag: false });
    out.push({ text: m[0], tag: true });
    last = m.index + m[0].length;
  }
  if (last < s.length) out.push({ text: s.slice(last), tag: false });
  return out;
}

export function initials(name) {
  const parts = String(name ?? "").trim().split(/\s+/).filter(Boolean);
  if (!parts.length) return "?";
  return (parts[0][0] + (parts.length > 1 ? parts[parts.length - 1][0] : "")).toUpperCase();
}

/** Local date (YYYY-MM-DD) of an instant in a time zone. */
export function localDate(iso, tz) {
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return null;
  const parts = new Intl.DateTimeFormat("en-CA", { timeZone: tz || "UTC", year: "numeric", month: "2-digit", day: "2-digit" })
    .formatToParts(d);
  const get = (t) => parts.find((p) => p.type === t)?.value;
  return `${get("year")}-${get("month")}-${get("day")}`;
}

export function localTime(iso, tz) {
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return null;
  return new Intl.DateTimeFormat("en-GB", { timeZone: tz || "UTC", hour: "2-digit", minute: "2-digit" }).format(d);
}

export function addDays(ymd, n) {
  const d = new Date(`${ymd}T12:00:00Z`);
  d.setUTCDate(d.getUTCDate() + n);
  return d.toISOString().slice(0, 10);
}

export function weekday(ymd) {
  return new Date(`${ymd}T12:00:00Z`).toLocaleDateString("en-GB", { weekday: "short", timeZone: "UTC" });
}

const APPROVAL_OF_STATE = { AWAITING_APPROVAL: "pending", APPROVED: "approved", READY_TO_PUBLISH: "approved",
  PUBLISHING: "approved", PUBLISHED: "approved", PUBLISH_FAILED: "approved", NEEDS_RECONCILE: "approved",
  REJECTED: "rejected" };
export function approvalOf(state) {
  return APPROVAL_OF_STATE[state] ?? "not yet";
}

/**
 * The next `days` days as buckets, merging the private plan (pipeline mirror),
 * scheduled cloud consents and free publishing slots. Nothing is inferred: an
 * item without a post says so.
 */
export function upcoming({ pipeline, cloud, now, tz, days = 7 }) {
  const today = localDate(now, tz);
  const end = addDays(today, days - 1);
  const posts = new Map((pipeline?.posts ?? []).map((p) => [p.post_id, p]));
  const cloudPosts = new Map((cloud?.posts ?? []).map((p) => [p.post_id, p]));
  const pubs = new Map((cloud?.publications ?? []).map((p) => [p.post_id, p]));
  const buckets = new Map();
  for (let i = 0; i < days; i++) buckets.set(addDays(today, i), []);
  const scheduled = new Set();
  for (const c of cloud?.consents ?? []) {
    if (c.status !== "active") continue;
    const day = localDate(c.slot_utc, tz);
    if (!buckets.has(day)) continue;
    const p = posts.get(c.post_id) ?? cloudPosts.get(c.post_id) ?? { post_id: c.post_id };
    const cp = cloudPosts.get(c.post_id);
    scheduled.add(c.post_id);
    buckets.get(day).push({ kind: "scheduled", date: day, time: localTime(c.slot_utc, tz), post_id: c.post_id,
      title: hook(p.text ?? cp?.text) ?? p.topic ?? c.post_id, type: contentType({ ...cp, ...p }),
      state: cp?.state ?? p.state ?? null, approval: "approved", publishing: pubs.get(c.post_id)?.state ?? "scheduled",
      slot_id: c.slot_id, consent_id: c.consent_id });
  }
  for (const e of pipeline?.calendar ?? []) {
    const day = String(e.date ?? "");
    if (!buckets.has(day) || (e.draft_ref && scheduled.has(e.draft_ref))) continue;
    const p = e.draft_ref ? posts.get(e.draft_ref) : null;
    buckets.get(day).push({ kind: "planned", date: day, time: null, post_id: e.draft_ref ?? null,
      title: (p && hook(p.text)) ?? e.topic ?? UNKNOWN, type: contentType({ format: e.format, ...(p ?? {}) }),
      state: p?.state ?? null, plan_status: e.status ?? null, approval: p ? approvalOf(p.state) : "no post yet",
      publishing: p ? (cloudPosts.has(p.post_id) ? "in cloud queue" : "not scheduled") : "no post yet" });
  }
  const taken = new Set([...(cloud?.consents ?? [])].filter((c) => c.status === "active").map((c) => c.slot_id));
  for (const s of cloud?.upcoming_slots ?? []) {
    const day = localDate(s.utc, tz);
    if (!buckets.has(day) || taken.has(s.slot_id)) continue;
    buckets.get(day).push({ kind: "slot", date: day, time: localTime(s.utc, tz), slot_id: s.slot_id, title: "Free slot" });
  }
  for (const list of buckets.values()) list.sort((a, b) => String(a.time ?? "99").localeCompare(String(b.time ?? "99")));
  return { from: today, to: end, days: [...buckets.entries()].map(([date, items]) => ({ date, items })) };
}

/** One chronological list from cloud events, decisions and private post histories (newest first). */
export function timeline({ cloud, pipeline, postId = null }) {
  const out = [];
  for (const e of cloud?.events ?? []) {
    let detail = {};
    try { detail = typeof e.detail === "string" ? JSON.parse(e.detail) : (e.detail ?? {}); } catch { detail = {}; }
    out.push({ at: e.at, stage: STAGE_OF_EVENT[e.event] ?? "other", label: e.event, post_id: e.post_id ?? null,
      id: detail.consent_id ?? detail.decision_id ?? detail.remote_id ?? null, actor: e.actor, source: "cloud" });
  }
  for (const p of cloud?.publications ?? []) {
    if (p.state === "published" && p.verified_by) {
      out.push({ at: p.published_at ?? p.updated_at, stage: "verified", label: `verified by ${p.verified_by}`,
        post_id: p.post_id, id: p.remote_id ?? p.idempotency_key ?? null, actor: "linkedin", source: "cloud" });
    }
  }
  for (const p of pipeline?.posts ?? []) {
    for (const h of p.history ?? []) {
      const stage = h.state === "APPROVED" ? "approved" : h.state === "READY_TO_PUBLISH" ? "approved"
        : ["SELECTED", "DRAFTED", "HUMANIZED", "QA_PASSED", "DUPLICATE_CHECKED", "AWAITING_APPROVAL"].includes(h.state) ? "planned"
          : h.state === "PUBLISHED" ? "executed" : h.state === "PUBLISH_FAILED" ? "failed"
            : h.state === "NEEDS_RECONCILE" ? "needs_reconcile" : "other";
      out.push({ at: h.at, stage, label: `${h.state}${h.note ? " — " + h.note : ""}`, post_id: p.post_id, id: null,
        actor: "pipeline", source: "git" });
    }
  }
  return out.filter((x) => x.at && (!postId || x.post_id === postId))
    .sort((a, b) => String(b.at).localeCompare(String(a.at)));
}

// a refusal the owner resolved by an override is history, not something that failed now
const OVERRIDDEN = /overridden by the owner/i;

/** Items that need the owner's attention, most urgent first. */
export function attention({ cloud, pipeline, decisions }) {
  const out = [];
  for (const p of cloud?.posts ?? []) {
    if (p.state === "NEEDS_RECONCILE" || p.state === "PUBLISHING") out.push({ level: "err", post_id: p.post_id, text: "Publication outcome unknown: reconcile" });
    if (p.state === "PUBLISH_FAILED") out.push({ level: "err", post_id: p.post_id, text: "Publication failed" });
  }
  const settled = new Set((pipeline?.posts ?? []).filter(isSettled).map((p) => p.post_id));
  for (const d of decisions ?? []) {
    if (d.status === "refused" && !settled.has(d.post_id) && !OVERRIDDEN.test(d.result ?? "")) out.push({ level: "warn", post_id: d.post_id, text: `Decision refused (${d.action}): ${d.result ?? ""}` });
  }
  for (const p of pipeline?.posts ?? []) {
    if (settled.has(p.post_id)) continue;   // history, not work
    if (p.state === "AWAITING_APPROVAL") out.push({ level: "info", post_id: p.post_id, text: "Ready for your approval" });
    if (p.refresh_request && p.plan_date && p.plan_date < new Date().toISOString().slice(0, 10)) {
      out.push({ level: "warn", post_id: p.post_id, text: "Slot passed: reschedule it to get the replacement written" });
    } else if (p.state === "NEEDS_REVISION") out.push({ level: "warn", post_id: p.post_id, text: "Needs revision" });
    if (p.state === "NEEDS_INPUT") out.push({ level: "warn", post_id: p.post_id, text: "Needs your input" });
  }
  const rank = { err: 0, warn: 1, info: 2 };
  return out.sort((a, b) => rank[a.level] - rank[b.level]);
}

/** Days until an ISO instant (floor), or null. */
export function daysUntil(iso, nowIso) {
  if (!iso) return null;
  const ms = Date.parse(iso) - Date.parse(nowIso);
  return Number.isNaN(ms) ? null : Math.floor(ms / 86400000);
}

export function parseHash(hash) {
  const h = String(hash ?? "").replace(/^#\/?/, "");
  const [view, ...rest] = h.split("/");
  return { view: view || "overview", arg: rest.length ? decodeURIComponent(rest.join("/")) : null };
}

/** Display status of one post/plan entry. Cloud facts (scheduled, published) win over the plan. */
export function displayStatus({ post = null, cloudPost = null, consent = null, publication = null, planStatus = null, pending = null } = {}) {
  const cs = cloudPost && cloudPost.state !== "WITHDRAWN" ? cloudPost.state : null;
  if (publication?.state === "published" || cs === "PUBLISHED" || post?.state === "PUBLISHED") return "PUBLISHED";
  if (cs === "NEEDS_RECONCILE" || cs === "PUBLISHING" || post?.state === "NEEDS_RECONCILE") return "NEEDS_RECONCILE";
  if (cs === "PUBLISH_FAILED") return "FAILED";
  if (consent) return "SCHEDULED";
  if (planStatus === "skipped") return "SKIPPED";
  if (planStatus === "open" && !post) return "OPEN_SLOT";     // LCE-049: reserved by the rolling calendar
  // LCE-043: a recorded Skip / Reject / Refresh takes effect for the owner at once; the repository
  // catches up at the next private run (the decision is pending there, never in doubt here).
  if (pending?.action === "skip") return "SKIPPED";
  if (pending?.action === "reject") return "REJECTED";
  if (pending?.action === "refresh") return "REPLACEMENT_PENDING";
  const st = post?.state;
  if (st === "REJECTED") return /^skipped/i.test(post?.approval?.reason ?? lastNote(post)) ? "SKIPPED" : "REJECTED";
  if (post?.refresh_request) return "REPLACEMENT_PENDING";     // LCE-042: rejected by Refresh, replacement coming
  if (st === "NEEDS_REVISION" || post?.regeneration) return "NEEDS_REGENERATION";
  if (cs === "READY_TO_PUBLISH" || st === "APPROVED" || st === "READY_TO_PUBLISH") return "APPROVED";
  if (st === "PUBLISH_FAILED") return "FAILED";
  if (st === "AWAITING_APPROVAL") return "AWAITING_APPROVAL";
  return "PLANNED";
}

// ── LCE-048: real Refresh progress, derived only from recorded events ─────────────────────
export const PROGRESS_STEPS = [
  { key: "recorded", label: "Request recorded" },
  { key: "dispatched", label: "Starting writer" },
  { key: "waiting", label: "Writer session started" },
  { key: "researching", label: "Researching" },
  { key: "writing", label: "Writing" },
  { key: "media", label: "Media" },
  { key: "humanization", label: "Humanization" },
  { key: "qa", label: "QA" },
  { key: "duplicate_check", label: "Duplicate check" },
  { key: "approval_prepared", label: "Approval prepared" },
  { key: "replacement_ready", label: "Ready" },
];
export const WAIT_STALL_MIN = 30;     // applied to the repository, yet no writing session has started
export const APPLY_STALL_MIN = 75;    // still not picked up by the private run (it runs hourly)
export const WORK_STALL_MIN = 20;     // a writing session started, then reported nothing more

const ms = (t) => (t ? Date.parse(t) : NaN);
export function fmtElapsed(sec) {
  sec = Math.max(0, Math.floor(sec));
  const h = Math.floor(sec / 3600), m = Math.floor((sec % 3600) / 60), s = sec % 60;
  const pad = (n) => String(n).padStart(2, "0");
  return h ? `${h}:${pad(m)}:${pad(s)}` : `${pad(m)}:${pad(s)}`;
}

/** The state of one post's Refresh from GET /api/refresh-status (never estimated). */
export function refreshProgress(st, nowMs = Date.now()) {
  const d = st?.decision;
  if (!d) return null;
  const post = st.mirror?.post ?? null;
  const ev = (st.events ?? []).filter((e) => !e.decision_id || e.decision_id === d.decision_id);
  const at = (stage) => ev.filter((e) => e.stage === stage).at(-1)?.at ?? null;
  const auto = d.origin === "freshness";   // LCE-049: requested by the freshness check, not by a click
  const ready = auto
    ? Boolean(!post?.refresh_request && post?.refresh?.requested_at === d.created_at)
    : Boolean(post?.refresh?.decision_id === d.decision_id && !post?.refresh_request) || Boolean(at("replacement_ready") && !post?.refresh_request);
  const failedAt = at("failed");
  const failed = !ready && failedAt && ev.at(-1)?.stage === "failed";
  const started = at("worker_started");
  const applied = d.status === "applied" ? d.resolved_at : null;
  const steps = PROGRESS_STEPS.map((s) => ({ ...s, at: null, state: "pending" }));
  const set = (key, t) => { const x = steps.find((s) => s.key === key); if (x && t) { x.at = t; x.state = "done"; } };
  set("recorded", d.created_at);
  const dispatched = ev.filter((e) => e.stage === "dispatched").at(-1) ?? null;
  const writerStarted = dispatched && !/^not started/.test(dispatched.note ?? "");
  if (writerStarted) set("dispatched", dispatched.at);
  set("waiting", started);
  for (const k of ["researching", "writing", "media", "humanization", "qa", "duplicate_check", "approval_prepared"]) set(k, at(k));
  if (ready) set("replacement_ready", at("replacement_ready") ?? post?.refresh?.completed_at ?? st.mirror?.generated_at);
  // a later stage implies the earlier ones happened; their own time is unknown (shown without one)
  const lastDone = steps.map((s) => s.state).lastIndexOf("done");
  steps.forEach((s, i) => { if (i < lastDone && s.state !== "done") s.state = "done_untimed"; });
  // design plan 03: while waiting for a writer, "Starting writer" is behind us (time unknown) and the
  // writer session is what we wait for, so the list never contradicts the phase shown above it
  if (!ready && !failed && !started && applied && !writerStarted) {
    const ds = steps.find((s) => s.key === "dispatched");
    if (ds.state === "pending") ds.state = "done_untimed";
  }
  const current = ready || failed ? null : steps.find((s) => s.state === "pending");
  if (current) current.state = "current";
  const lastEvent = ev.at(-1)?.at ?? null;
  const mins = (t) => (nowMs - ms(t)) / 60000;
  let phase, note = null, stalled = false;
  if (ready) phase = "ready";
  else if (failed) { phase = "failed"; note = ev.at(-1)?.note ?? null; }
  else if (!started && writerStarted) {   // LCE-050: the writer was started at once; its session is coming up
    phase = "starting";
    note = "The writer routine was started for this request; its session is setting up.";
    if (mins(dispatched.at) > WAIT_STALL_MIN) { stalled = true; note = `The writer was started ${Math.round(mins(dispatched.at))} minutes ago but has not reported yet.`; }
  } else if (!applied && !started) {
    phase = "queued";
    note = dispatched ? `Writer ${dispatched.note}.` : "Recorded in the cloud; the private run adds it to the repository (it runs hourly).";
    if (mins(d.created_at) > APPLY_STALL_MIN) { stalled = true; note = `Not picked up by the private run after ${Math.round(mins(d.created_at))} minutes.`; }
  } else if (!started) {
    phase = "waiting";
    note = dispatched && !writerStarted ? `In the repository. Writer ${dispatched.note}.` : "In the repository; no writing session has started yet.";
    if (applied && mins(applied) > WAIT_STALL_MIN) { stalled = true; note = `Waiting for writing session — no worker has started yet (${Math.round(mins(applied))} min since it reached the repository).`; }
  } else {
    phase = "working";
    note = ev.at(-1)?.note ? `Last report: ${ev.at(-1).note}.` : null;
    if (lastEvent && mins(lastEvent) > WORK_STALL_MIN) {
      stalled = true;
      note = `The writing session has reported nothing for ${Math.round(mins(lastEvent))} minutes (last: ${ev.at(-1).stage}).`;
    }
  }
  if (auto) steps[0].label = "Found stale by the freshness check";
  return { decision_id: d.decision_id, auto, phase, stalled, note, steps, ready, failed: Boolean(failed),
    started_at: d.created_at, elapsed: fmtElapsed(((ready || failed ? ms(lastEvent ?? d.created_at) : nowMs) - ms(d.created_at)) / 1000),
    events: ev };
}

/** LCE-047: how long the Access session Cloudflare issued actually lasts (from the token's own iat/exp,
 * reported by /api/whoami), in honest units, and whether it is too short to use the Control Center. */
export function sessionLength(w) {
  if (w?.session_issued_at == null || w?.session_expires_at == null) return null;
  const s = w.session_expires_at - w.session_issued_at;
  const text = s < 120 ? `${s} seconds` : s < 7200 ? `${Math.round(s / 60)} minutes` : `${Math.round(s / 3600)} hours`;
  return { seconds: s, text, tooShort: s < 300 };
}

/** LCE-045: what the owner's last action here amounts to NOW. A "recorded" click is only shown as
 * recorded while D1 still has that decision pending; once resolved, its real fate is shown instead
 * (a page left open must never claim a decision that was later refused or replaced is in effect). */
export function outcomeNow(o, decisions = []) {
  if (!o || o.state !== "recorded" || !o.decision_id) return o ? { kind: o.state } : null;
  const d = (decisions ?? []).find((x) => x.decision_id === o.decision_id);
  if (!d) return { kind: "recorded" };   // not in the snapshot (yet): nothing proves otherwise
  if (d.status === "pending") return { kind: "recorded" };
  const fate = {
    applied: "applied to the repository",
    refused: `not applied${d.result ? ": " + d.result : ""}`,
    cancelled: "cancelled (undone)",
    superseded: "replaced by a later decision",
  }[d.status] ?? d.status;
  return { kind: "resolved", status: d.status, text: fate, at: d.resolved_at ?? null };
}

/** LCE-043: decisions that end this version as a candidate. While one is pending, nothing may act on the
 * version (no Approve, Refresh, Edit, Reschedule, Skip or Reject); only Cancel undoes it. */
export const CLOSING = new Set(["skip", "reject", "refresh"]);
export const closedBy = (pending) => (pending && pending.status === "pending" && CLOSING.has(pending.action) ? pending.action : null);

/** The pending decision for a post or an empty plan entry (cloud snapshot, newest first). */
export function pendingDecision(decisions, postId, planDate = null) {
  return (decisions ?? []).find((d) => d.status === "pending" &&
    (postId ? d.post_id === postId : !d.post_id && planDate && d.plan_date === planDate)) ?? null;
}

const lastNote = (post) => String((post?.history ?? []).at(-1)?.note ?? "");

export function draftingNote(post) {
  if (!post) return "No post written yet";
  if (post.state === "NEEDS_INPUT") return "Needs your input";
  if (DRAFTING.has(post.state)) return "In writing";
  return null;
}

/** Media of a post: the type, and whether a real preview image exists (never a fake). */
export function media(post, cloud) {
  const cp = (cloud?.posts ?? []).find((p) => p.post_id === post?.post_id && p.state !== "WITHDRAWN");
  const type = contentType({ ...(cp ?? {}), ...(post ?? {}), image: post?.image ?? cp?.image });
  if (type.key !== "image") return { type, preview: "none" };
  if (cp?.image) return { type, preview: "available", source: "approved image", alt: post?.image?.alt_text ?? null };
  const pm = (cloud?.preview_media ?? []).find((m) => m.post_id === post?.post_id);
  if (pm && (!post?.image?.sha256 || pm.sha256 === post.image.sha256)) {
    return { type, preview: "available", source: "pipeline image", alt: post?.image?.alt_text ?? pm.alt_text ?? null };
  }
  return { type, preview: "missing", file: post?.image?.file ?? null, alt: post?.image?.alt_text ?? null };
}

/**
 * The content plan: every planned entry and post (from the private pipeline,
 * the source of truth), merged with cloud facts (scheduled, published) and
 * free publishing slots. A free slot never hides a planned post.
 */
export function contentPlan({ pipeline, cloud, now, tz, days = 14 }) {
  const today = localDate(now, tz);
  const end = addDays(today, days - 1);
  const posts = new Map((pipeline?.posts ?? []).map((p) => [p.post_id, p]));
  const cloudPosts = new Map((cloud?.posts ?? []).map((p) => [p.post_id, p]));
  const pubs = new Map((cloud?.publications ?? []).map((p) => [p.post_id, p]));
  const consents = new Map((cloud?.consents ?? []).filter((c) => c.status === "active").map((c) => [c.post_id, c]));
  const slots = (cloud?.upcoming_slots ?? []).map((s) => ({ ...s, date: localDate(s.utc, tz), time: localTime(s.utc, tz) }));
  const usedSlots = new Set([...consents.values()].map((c) => c.slot_id));
  const seen = new Set();
  const items = [];
  const add = (postId, entry) => {
    const post = postId ? posts.get(postId) ?? null : null;
    const cp = postId ? cloudPosts.get(postId) ?? null : null;
    const consent = postId ? consents.get(postId) ?? null : null;
    const pub = postId ? pubs.get(postId) ?? null : null;
    const pending = pendingDecision(cloud?.decisions, postId, postId ? null : entry?.date);
    const status = displayStatus({ post, cloudPost: cp, consent, publication: pub, planStatus: entry?.status, pending });
    let date = consent ? localDate(consent.slot_utc, tz) : String(entry?.date ?? post?.plan_date ?? cp?.plan_date ?? "") || null;
    if (status === "PUBLISHED" && pub?.published_at) date = localDate(pub.published_at, tz);
    let time = consent ? localTime(consent.slot_utc, tz) : status === "PUBLISHED" && pub?.published_at ? localTime(pub.published_at, tz) : null;
    let timeSource = consent ? "scheduled" : time ? "published" : null;
    if (!time && date) {
      const slot = slots.find((s) => s.date === date && !usedSlots.has(s.slot_id));
      if (slot) { time = slot.time; timeSource = "planned slot"; usedSlots.add(slot.slot_id); }
    }
    const text = post?.text ?? cp?.text ?? null;
    items.push({ kind: "post", post_id: postId, date, time, time_source: timeSource, status,
      title: hook(text) ?? entry?.topic ?? post?.topic ?? postId ?? UNKNOWN, topic: entry?.topic ?? post?.topic ?? null,
      objective: post?.objective ?? entry?.objective ?? null, media: media(post ?? cp ?? { post_id: postId, format: entry?.format }, cloud),
      note: closedBy(pending) ? "recorded; the repository is updated at the next private run"
        : status === "OPEN_SLOT" ? `Reserved by the rolling calendar (pillar ${entry?.pillar ?? "?"}); the writing Routine prepares a candidate, which then needs your approval`
          : draftingNote(post ?? (cp ? { state: "APPROVED" } : null)),
      pending: pending ? { decision_id: pending.decision_id, action: pending.action, created_at: pending.created_at, post_id: pending.post_id ?? null, plan_date: pending.plan_date ?? null } : null, consent_id: consent?.consent_id ?? null,
      plan_date: entry?.date ?? post?.plan_date ?? null, sources: post?.sources ?? (entry?.sources ?? []).map((u) => ({ url: u })) });
  };
  // An archived post (and its plan entry) is history: it leaves the plan, never the History view.
  for (const p of pipeline?.posts ?? []) if (isArchived(p)) seen.add(p.post_id);
  for (const e of pipeline?.calendar ?? []) {
    if (e.draft_ref && seen.has(e.draft_ref)) continue;
    if (e.draft_ref) seen.add(e.draft_ref);
    add(e.draft_ref ?? null, e);
  }
  for (const p of pipeline?.posts ?? []) if (!seen.has(p.post_id)) { seen.add(p.post_id); add(p.post_id, null); }
  for (const c of cloud?.posts ?? []) if (!seen.has(c.post_id)) { seen.add(c.post_id); add(c.post_id, null); }
  const closed = new Set(["PUBLISHED", "REJECTED", "SKIPPED"]);
  const pastDue = items.filter((i) => i.date && i.date < today && !closed.has(i.status))
    .sort((a, b) => a.date.localeCompare(b.date));
  const recent = items.filter((i) => i.status === "PUBLISHED").sort((a, b) => String(b.date).localeCompare(String(a.date)));
  const inRange = (i) => i.date && i.date >= today && i.date <= end;
  const byDay = [];
  for (let n = 0; n < days; n++) {
    const d = addDays(today, n);
    const dayItems = items.filter((i) => i.date === d);
    for (const s of slots) {
      if (s.date === d && !usedSlots.has(s.slot_id)) dayItems.push({ kind: "slot", date: d, time: s.time, slot_id: s.slot_id, local: s.local });
    }
    dayItems.sort((a, b) => String(a.time ?? "99:99").localeCompare(String(b.time ?? "99:99")));
    byDay.push({ date: d, items: dayItems });
  }
  const counts = Object.fromEntries(Object.keys(STATUSES).map((k) => [k, items.filter((i) => inRange(i) && i.status === k).length]));
  const undated = items.filter((i) => !i.date && !closed.has(i.status));
  return { from: today, to: end, items, days: byDay, pastDue, undated, recent, counts };
}

export function applyFilter(item, filterId) {
  if (!filterId || filterId === "all") return true;
  if (item.kind === "slot") return false;
  return Boolean(FILTERS.find((f) => f.id === filterId)?.match?.includes(item.status));
}

/**
 * Humanization status rows. Claims only what the post's recorded humanization
 * proves (LCE-037): profile file versions, objective, machine-checked rules.
 */
export function humanizationRows(post, voice) {
  const hz = post?.humanization;
  const rows = [];
  if (!hz) {
    rows.push({ label: "Voice profile", value: "not recorded", tone: "muted",
      note: "Written before humanization records existed: profile use cannot be verified." });
  } else {
    rows.push({ label: "Voice profile", value: hz.source === "owner_edit" ? "owner edit" : `applied (v${hz.voice_version ?? "?"})`,
      tone: hz.source === "owner_edit" ? "info" : hz.profile_current === false ? "warn" : "ok",
      note: hz.source === "owner_edit" ? "Text written by you in the Control Center."
        : hz.profile_current === false ? "Profile files changed since this text was written." : "Text recorded against the current voice, profile and brand files." });
    const c = hz.checklist ?? {};
    rows.push({ label: "Voice rules (machine-checked)", value: c.failed ? `${c.failed} failed` : `${c.passed ?? 0} passed`,
      tone: c.failed ? "err" : "ok", note: "Hashtags, emoji, avoided phrases, AI patterns, evidence, individual voice, format." });
  }
  const objectives = new Map((voice?.objectives ?? []).map((o) => [o.id, o.label]));
  const obj = post?.objective ?? hz?.objective ?? null;
  rows.push({ label: "Objective", value: obj ? objectives.get(obj) ?? obj : "not recorded",
    tone: obj ? "ok" : "muted", note: obj ? null : "No objective in the data for this post." });
  rows.push({ label: "Tone & positioning", value: "your review", tone: "muted", note: "Not machine-verifiable; judged at approval." });
  rows.push({ label: "QA", value: post?.qa?.status ?? "not run", tone: post?.qa?.status === "passed" ? "ok" : post?.qa?.status ? "err" : "muted" });
  rows.push({ label: "Duplicate check", value: post?.duplicate?.status ?? "not run",
    tone: post?.duplicate?.status === "passed" ? "ok" : post?.duplicate?.status ? "err" : "muted" });
  return rows;
}

// ── LCE-038: media record shown per post (from image.yaml via the mirror) ──
const ORIGIN_LABEL = { own_creation: "Own creation", owner_screenshot: "Your screenshot", owner_photo: "Your photo",
  source_publication: "Source publication", licensed_stock: "Licensed (Wikimedia Commons or stock)", generated: "Generated" };
const REASON_LABEL = { text_carries_point: "The text carries the point", no_relevant_visual: "No relevant visual",
  no_rights_safe_source: "No rights-safe source", personal_story_without_owner_photo: "Personal story without your photo",
  would_be_decorative: "An image would only decorate", no_suitable_licensed_image: "No suitable licensed image" };
const STATUS_LABEL = { attached: "Attached", text_only: "Text-only (decided)", needs_review: "Rights need your review",
  invalid: "Invalid — fix before approval", undecided: "No decision yet" };

export function mediaRecord(post) {
  const img = post?.image ?? null;
  if (!img) {
    return { status: "undecided", statusLabel: post ? "No media decision recorded" : STATUS_LABEL.undecided, rows: [],
      note: "This post has no media decision (it predates the media stage or the decision is still open)." };
  }
  const status = img.media_status ?? (img.kind === "none" ? "text_only" : "attached");
  const rows = [];
  const sv = img.source_visual;   // LCE-046: the post's own source was checked for its visual first
  if (sv) {
    rows.push(["Source visual", { source_visual_used: "used (reuse permitted)", source_visual_unavailable_or_restricted: "source_visual_unavailable_or_restricted",
      no_source_visual: "the source has no visual" }[sv.status] ?? sv.status]);
    if (sv.source_url) rows.push(["Source checked", { href: sv.source_url }]);
    if (sv.evidence) rows.push(["Source evidence", sv.evidence]);
  }
  if (img.kind === "none") {
    rows.push(["Reason", img.text_only_reason ? REASON_LABEL[img.text_only_reason] ?? img.text_only_reason : "not categorised"]);
    if (img.rationale) rows.push(["Rationale", img.rationale]);
  } else {
    const prov = img.provenance ?? {};
    rows.push(["Kind", img.kind]);
    rows.push(["Source", ORIGIN_LABEL[prov.origin] ?? prov.origin ?? UNKNOWN]);
    if (prov.source_name) rows.push(["Found via", prov.source_name]);
    if (prov.source_id) rows.push(["Source ID", prov.source_id]);
    if (prov.title) rows.push(["Source file", prov.title]);
    if (prov.creator) rows.push(["Creator", prov.creator]);
    else if (prov.credit) rows.push(["Credit", prov.credit]);
    if (prov.license || prov.usage) rows.push(["Rights", [prov.license, prov.usage].filter(Boolean).join(" · ")]);
    if (prov.license_url) rows.push(["Licence", { href: prov.license_url }]);
    if (prov.attribution_required !== undefined) rows.push(["Attribution", `${prov.attribution_required ? "required" : "not required"}${prov.attribution ? ` · “${prov.attribution}”` : ""} · kept in the image record, not shown in the post${prov.attribution_display ? ` (${prov.attribution_display})` : ""}`]);
    if (prov.source_url) rows.push(["Source URL", { href: prov.source_url }]);
    if (prov.retrieved) rows.push(["Retrieved", `${prov.retrieved}${prov.retrieved_at ? " · " + String(prov.retrieved_at).slice(0, 16).replace("T", " ") : ""}`]);
    if (prov.original_sha1) rows.push(["Original SHA-1", prov.original_sha1]);
    if (prov.generation?.method) rows.push(["Made with", prov.generation.method]);
    if (img.width && img.height) rows.push(["Size", `${img.width}×${img.height} px${img.bytes ? ` · ${Math.round(img.bytes / 1024)} KB` : ""}${img.mime ? ` · ${img.mime.replace("image/", "").toUpperCase()}` : ""}`]);
    else if (img.bytes) rows.push(["Size", `${Math.round(img.bytes / 1024)} KB`]);
    if (img.alt_text) rows.push(["Alt text", img.alt_text]);
    if (img.relation) rows.push(["Relation", img.relation]);
  }
  if (img.decided_by) rows.push(["Decided", `${img.decided_by}${img.decided_at ? " · " + String(img.decided_at).slice(0, 10) : ""}`]);
  return { status, statusLabel: STATUS_LABEL[status] ?? status, rows, errors: img.errors ?? [], warnings: img.warnings ?? [] };
}

export function mediaSourceShort(post) {
  const img = post?.image;
  if (!img || img.kind === "none") return null;
  const prov = img.provenance ?? {};

// ── LCE-040: same-day freshness (from posts/<id>/freshness.yaml via the mirror) ──
export const FRESHNESS = {
  not_checked: { label: "Not checked", tone: "muted" },
  checked_today: { label: "Checked today", tone: "warn" },
  no_changes: { label: "No changes needed", tone: "ok" },
  update_needed: { label: "Update needed", tone: "err" },
  updated_today: { label: "Updated today", tone: "info" },
  update_requires_approval: { label: "Update requires approval", tone: "warn" },
};

/**
 * Freshness of one post for `today` (local date): the latest refresh record,
 * whether it applies to the CURRENT text, and whether approval is still valid.
 * Nothing is assumed: no record (or one for another day or text) is "Not checked".
 */
export function freshness(post, { today, cloudRow = null, cloudPost = null, slotDate = null } = {}) {
  const latest = post?.freshness?.latest ?? null;
  const current = post?.actual_hash ?? null;
  const ap = post?.approval ?? {};
  let approval = "not approved";
  if (ap.state === "approved") approval = ap.approved_hash && current && ap.approved_hash !== current ? "invalid (text changed)" : "valid";
  else if (cloudPost?.approved_hash && cloudPost.state !== "WITHDRAWN") approval = "valid";
  else if (post?.state === "AWAITING_APPROVAL") approval = "awaiting approval";
  let key = "not_checked";
  let note = latest ? `last check ${latest.check_date}` : "no same-day check recorded";
  const forToday = latest && latest.check_date === today && !latest.test_mode;
  const sameText = latest && (!current || latest.content_hash === current);
  if (latest?.test_mode) {
    note = `test run as of ${latest.check_date} (not a same-day check)`;
  } else if (forToday && !sameText) {
    note = "checked an earlier version of the text";
  } else if (forToday) {
    note = latest.reason || null;
    if (latest.decision === "updated") key = latest.status === "update_awaiting_approval" ? "update_requires_approval" : "updated_today";
    else if (latest.decision === "update_required" || latest.status === "update_required") key = "update_needed";
    else if (latest.status === "current") key = "no_changes";
    else key = "checked_today";
  }
  if (latest?.approval_effect === "invalidated" && forToday) approval = post?.state === "AWAITING_APPROVAL" ? "invalidated by refresh — approve the new version" : "invalidated by refresh";
  // What the Worker's scheduled-publish gate sees (D1 freshness row, LCE-040).
  let gate = null;
  if (slotDate) {
    const hash = cloudPost?.approved_hash ?? null;
    if (!cloudRow) gate = { ok: false, why: "no fresh check received" };
    else if (cloudRow.check_date !== slotDate) gate = { ok: false, why: `last check ${cloudRow.check_date}, not the publishing day` };
    else if (cloudRow.status !== "current") gate = { ok: false, why: `check status ${cloudRow.status}` };
    else if (hash && cloudRow.content_hash !== hash) gate = { ok: false, why: "check was for another text version" };
    else gate = { ok: true, why: "same-day check accepted" };
  }
  return { key, ...FRESHNESS[key], note, approval, latest, history: post?.freshness?.history ?? [], gate };
}

// ── LCE-041: manual refresh of the whole post package ──
export const REFRESH_CONFIRM = "This rejects the current version and generates a completely new replacement for the same slot: " +
  "new text, sources and a new image. The current version will be preserved in History and the replacement will require " +
  "your approval.";
export const SKIP_CONFIRM = "This post will not be published and its slot is released. No replacement is generated " +
  "(use Refresh for a replacement).";
const UNPUBLISHED_CLOSED = new Set(["PUBLISHED", "PUBLISHING", "PUBLISH_FAILED", "NEEDS_RECONCILE", "REJECTED"]);

/** Can the owner ask for a refresh? Only unpublished posts that are not in the cloud publish queue. */
export function canRefresh(post, inCloud) {
  return Boolean(post && post.text && !inCloud && !post.refresh_request && !UNPUBLISHED_CLOSED.has(post.state));
}

/** Where a refresh stands for this post (from the recorded request/outcome only). */
export function refreshState(post, pending = null) {
  if (pending?.action === "skip" || pending?.action === "reject") return null;   // LCE-043: the version is closed
  if (pending?.action === "refresh") {
    return { key: "recorded", label: "Refresh requested", tone: "info",
      note: "Recorded in the cloud; the private workflow adds it to the repository within about an hour." };
  }
  if (post?.refresh_request?.origin === "freshness") {   // LCE-049
    const rr = post.refresh_request, v = rr.rejected_version;
    return { key: "stale", label: "Sources changed, writing an update", tone: "info", auto: true,
      note: `The freshness check before this slot found: ${rr.stale?.reason ?? "the content is out of date"}. ` +
        `${v ? `The earlier version is kept as v${v} and its approval no longer applies. ` : ""}` +
        "A writing session is producing an updated post for the same slot; it will need your approval." };
  }
  if (post?.refresh_request) {
    const v = post.refresh_request.rejected_version;
    return { key: "requested", label: "Refresh requested", tone: "info",
      note: `${v ? `The rejected version is kept as v${v}. ` : ""}A writing session (the scheduled Routine, every 2 hours, or a manual session) is producing a new replacement; it will need your approval.` };
  }
  const r = post?.refresh;
  if (r?.outcome === "refreshed" && r.origin === "freshness" && post.state === "AWAITING_APPROVAL") {
    return { key: "refreshed", label: "Updated for current sources, ready for approval", tone: "warn", note: r.reason ?? null };
  }
  if (r?.outcome === "refreshed" && post.state === "AWAITING_APPROVAL") {
    return { key: "refreshed", label: "Refreshed, ready for approval", tone: "warn", note: r.reason ?? null };
  }
  if (r?.outcome === "refreshed") return { key: "refreshed_done", label: "Refreshed", tone: "muted", note: r.reason ?? null };
  if (r?.outcome === "kept") return { key: "kept", label: "Refresh checked, current version kept", tone: "ok", note: r.reason ?? null };
  return null;
}

/** The media relevance record as rows (never invented: missing → said so). */
export function relevanceRows(image) {
  const rel = image?.media_relevance;
  if (!image || image.kind === "none") return { rows: [], decision: null };
  if (!rel) return { rows: [["Relevance", "not recorded"]], decision: "missing" };
  const pct = (x) => (x === null || x === undefined ? "not machine-checked (declared)" : `${Math.round(x * 100)}%`);
  const rows = [
    ["Visual concept", rel.concept ?? "not stated"],
    ["Visual type", rel.visual_type ?? "not stated"],
    ["Why it helps", rel.relevance_reason ?? "not stated"],
    ["Copied post text", pct(rel.copied_post_text_ratio)],
    ["Facts in the image", (rel.factual_claims ?? []).length
      ? rel.factual_claims.map((f) => `${f.text}${f.supported ? " (sourced)" : " (UNSUPPORTED)"}`).join("; ") : "none"],
  ];
  if ((rel.source_requirements ?? []).length) rows.push(["Source shown", rel.source_requirements.join("; ")]);
  const sem = rel.semantic;   // LCE-043: a real image is relevant when its own metadata names the post's subject
  if (sem?.depicts) {   // LCE-044: chosen after looking at the image
    rows.push(["Depicts", sem.depicts]);
    if (sem.association) rows.push(["Tied to the source", `${sem.association.replaceAll("_", " ")}${sem.why_belongs_to_source ? " — " + sem.why_belongs_to_source : ""}`]);
    if (sem.why_legal) rows.push(["Why reuse is legal", sem.why_legal]);
    rows.push(["Why this image", sem.why_relevant ?? "not stated"]);
    rows.push(["Reviewed", `${sem.reviewed_by ?? "not stated"} · ${sem.method ?? "visual review"}`]);
  } else if (sem) {
    rows.push(["Subject", sem.subject ?? "not stated"]);
    rows.push(["In source metadata", (sem.matched_terms ?? []).length ? sem.matched_terms.join(", ") : "none (not relevant)"]);
  }
  return { rows, decision: rel.media_decision, legacy: Boolean(rel.legacy), problems: rel.problems ?? [] };
}

/** Previous → current for the History view: the last preserved version and the current package. */
export function versionCompare(post, versionMedia = []) {
  const vs = post?.versions ?? [];
  if (!vs.length) return null;
  const prev = vs[vs.length - 1];
  const hasImg = (v) => versionMedia.some((m) => m.post_id === post.post_id && Number(m.version) === v.version);
  return {
    previous: { ...prev, image_available: Boolean(prev.image_file) && hasImg(prev) },
    current: { hook: hook(post.text), content_hash: post.actual_hash, image_sha256: post.image?.sha256 ?? null,
      concept: post.image?.media_relevance?.concept ?? null, state: post.state, refresh: post.refresh ?? null },
    older: vs.slice(0, -1).reverse().map((v) => ({ ...v, image_available: Boolean(v.image_file) && hasImg(v) })),
  };
}


/** Owner request 2026-10-06: a post with nothing left to do. A rejected or published post, or one the owner
 * archived, is history: its old refresh failures and refused decisions no longer ask for attention, and an
 * archived post leaves Overview and Upcoming (History keeps it, with every record). */
export const SETTLED_STATES = new Set(["REJECTED", "PUBLISHED"]);
export const isArchived = (post) => Boolean(post?.archived);
export const isSettled = (post) => Boolean(post) && (isArchived(post) || SETTLED_STATES.has(post.state));
export const canArchive = (post, cloudPost) => Boolean(post) && !isArchived(post) && SETTLED_STATES.has(post.state)
  && !(cloudPost && ["READY_TO_PUBLISH", "PUBLISHING", "PUBLISH_FAILED", "NEEDS_RECONCILE"].includes(cloudPost.state));

// ── LCE-050: the "right now" board and the Content Radar ───────────────────────────────
/** What needs the owner, and what the system is doing, from recorded facts only. */
export function nowBoard({ pipeline, cloud, progress = {}, nowMs = Date.now() }) {
  const posts = pipeline?.posts ?? [];
  const settled = new Set(posts.filter(isSettled).map((p) => p.post_id));
  const pending = (cloud?.decisions ?? []).filter((d) => d.status === "pending");
  const closing = new Set(pending.filter((d) => ["skip", "reject", "refresh"].includes(d.action)).map((d) => d.post_id));
  const work = pipeline?.freshness_plan?.work ?? [];
  const today = new Date(nowMs).toISOString().slice(0, 10);
  const lastStage = (id) => (progress[id]?.events ?? []).at(-1)?.stage ?? null;
  const approval = posts.filter((p) => p.state === "AWAITING_APPROVAL" && !closing.has(p.post_id));
  const researching = work.filter((u) => u.kind === "research" || (u.post_id && lastStage(u.post_id) === "researching"));
  const writing = [
    ...posts.filter((p) => !settled.has(p.post_id)).filter((p) => (p.refresh_request && !(p.plan_date && p.plan_date < today)) || pending.some((d) => d.action === "refresh" && d.post_id === p.post_id))
      .map((p) => ({ post_id: p.post_id, plan_date: p.plan_date, kind: "replacement" })),
    ...work.filter((u) => u.kind === "slot" && u.started),
  ];
  const failed = [
    ...(cloud?.posts ?? []).filter((p) => ["PUBLISH_FAILED", "NEEDS_RECONCILE"].includes(p.state)).map((p) => ({ post_id: p.post_id, why: p.state })),
    ...(cloud?.decisions ?? []).filter((d) => d.status === "refused" && !settled.has(d.post_id) && !OVERRIDDEN.test(d.result ?? "") && Date.parse(d.created_at) > nowMs - 7 * 86400e3).map((d) => ({ post_id: d.post_id, why: `${d.action} refused` })),
    ...Object.entries(progress).filter(([id, st]) => !settled.has(id) && (st?.events ?? []).at(-1)?.stage === "failed").map(([id, st]) => ({ post_id: id, why: st.events.at(-1).note ?? "refresh failed" })),
  ];
  const radar = pipeline?.radar ?? {};
  const fresh = (radar.items ?? []).filter((i) => Date.parse(i.published_at ?? i.first_seen) > nowMs - 86400e3);
  const scheduled = (cloud?.consents ?? []).filter((c) => c.status === "active");
  const next = (pipeline?.calendar ?? []).filter((e) => String(e.date) >= today && !["skipped", "rejected", "published"].includes(e.status))
    .sort((a, b) => String(a.date).localeCompare(String(b.date)))[0] ?? null;
  return { approval, researching, writing, fresh, scheduled, failed, next,
    radarConfigured: Boolean(radar.configured), radarNewToday: radar.new_today ?? fresh.length };
}

/** Radar items for the page: filter by pillar and use, newest first (never invented fields). */
export function radarRows(radar, { pillar = "all", unusedOnly = false, order = "rank" } = {}) {
  // the mirror lists items in rank order (relevance, freshness, source quality); "newest" re-sorts by date
  const rows = (radar?.items ?? [])
    .filter((i) => pillar === "all" || i.pillar === pillar)
    .filter((i) => !unusedOnly || !i.used_by);
  return order === "newest"
    ? rows.slice().sort((a, b) => String(b.published_at ?? b.first_seen).localeCompare(String(a.published_at ?? a.first_seen)))
    : rows;
}

export function fmtAge(hours) {
  if (hours == null) return "date unknown";
  if (hours < 1) return "under an hour ago";
  if (hours < 48) return `${Math.round(hours)} h ago`;
  return `${Math.round(hours / 24)} days ago`;
}
