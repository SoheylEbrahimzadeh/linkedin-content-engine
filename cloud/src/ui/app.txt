function h(tag, attrs, ...kids) {
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs || {})) {
    if (v === null || v === undefined || v === false) continue;
    if (k === "class") el.className = v;
    else if (k.startsWith("on")) el.addEventListener(k.slice(2), v);
    else el.setAttribute(k, v === true ? "" : String(v));
  }
  for (const c of kids.flat(Infinity)) {
    if (c === null || c === undefined || c === false) continue;
    el.append(c instanceof Node ? c : document.createTextNode(String(c)));
  }
  return el;
}

// Cloudflare Access answers an expired session with a redirect to its login page (another origin).
// fetch() would then fail with a bare "Failed to fetch"; with redirect: "manual" we can say what happened.
// ── LCE-042: reliable actions ─────────────────────────────────────────
// Cloudflare Access answers a request whose session it does not accept with a redirect to its sign-in
// page (another origin). With redirect: "manual" the page can tell that apart from a Worker answer, keep
// the action, sign in again in a small top-level window, and send the SAME action once more (same
// request_id, so the Worker never records it twice). What the browser saw is reported for diagnosis.

function sessionNote() {
  const w = S.whoami;
  if (!w?.session_expires_at) return "";
  const end = new Date(w.session_expires_at * 1000).toISOString();
  const len = sessionLength(w);
  return ` Your Access session ${w.session_expires_at * 1000 < Date.now() ? "ended" : "ends"} at ${localTime(end, tz())}` +
    (len ? ` (session length ${len.text}).` : ".") +
    (len?.tooShort ? " " + SHORT_SESSION_ADVICE.replace("%s", len.text) : "");
}

/** One request; never throws. kind: ok | signin (Access redirect) | network | http. */
async function rawFetch(method, path, body) {
  let res;
  try {
    res = await fetch("/api" + path, {
      method, credentials: "same-origin", cache: "no-store", redirect: "manual",
      headers: { "content-type": "application/json", "x-lce-client": "dashboard" },
      body: body === undefined ? undefined : JSON.stringify(body),
    });
  } catch (e) {
    return { kind: "network", error: String(e?.message || e) };
  }
  if (res.type === "opaqueredirect" || res.status === 0) return { kind: "signin", status: res.status, type: res.type };
  let data = {};
  try { data = await res.json(); } catch { data = {}; }
  if (!res.ok) return { kind: "http", status: res.status, data, error: data.error || data.detail || data.reason || ("HTTP " + res.status) };
  if (method === "GET") T.lastOkGetAt = new Date().toISOString();
  return { kind: "ok", status: res.status, data };
}

async function api(method, path, body) {
  const r = await rawFetch(method, path, body);
  if (r.kind === "ok") return r.data;
  const err = new Error(r.kind === "signin" ? SIGNIN + sessionNote()
    : r.kind === "network" ? `the Control Center could not be reached (${r.error})` : r.error);
  err.status = r.status; err.signin = r.kind === "signin"; err.data = r.data;
  throw err;
}

// A sign-in window: Access redirects top-level pages through its login and back; the window then tells us.
function signInWindow() {
  return new Promise((resolve) => {
    const w = window.open("/?reauth=1", "lce-reauth", "width=520,height=680");
    if (!w) { resolve(false); return; }
    let done = false;
    const finish = (ok) => { if (done) return; done = true; window.removeEventListener("message", onMsg); clearInterval(t); resolve(ok); };
    const onMsg = (e) => { if (e.origin === location.origin && e.data === "lce-reauth-ok") finish(true); };
    window.addEventListener("message", onMsg);
    const t = setInterval(() => { if (w.closed) setTimeout(() => finish(false), 400); }, 500);
  });
}

// Keeps the action while asking to sign in: tries the window at once (still inside the click), otherwise
// a banner button opens it. Resolves true once signed in, false if the owner closes the window.
async function signInKeeping(label) {
  if (await signInWindow()) { document.getElementById("session-banner")?.remove(); return true; }
  return new Promise((resolve) => {
    document.getElementById("session-banner")?.remove();
    const bar = h("div", { id: "session-banner", class: "notice warn session-banner", role: "alert" },
      h("strong", {}, "Sign-in needed. "), `${SIGNIN}${sessionNote()} Your ${label} is kept and will be sent once, right after you sign in. `,
      h("button", { type: "button", class: "small primary", onclick: async () => {
        const ok = await signInWindow();
        if (ok) { bar.remove(); resolve(true); }
      } }, `Sign in and send the ${label}`),
      h("button", { type: "button", class: "small", onclick: () => { bar.remove(); resolve(false); } }, "Cancel"));
    document.body.prepend(bar);
  });
}

async function refreshWhoami() {
  const r = await rawFetch("GET", "/whoami");
  if (r.kind === "ok") S.whoami = r.data;
  return r.kind;
}

function evidence(kind, extra) {
  const w = S.whoami ?? {};
  reports.push({ kind, detail: { ...extra, page_loaded_at: T.pageLoadedAt, last_ok_get_at: T.lastOkGetAt,
    session_issued_at: w.session_issued_at ?? null, session_expires_at: w.session_expires_at ?? null,
    browser_now: new Date().toISOString(), user_agent: navigator.userAgent.slice(0, 160) } });
}
async function flushReports() {
  while (reports.length) {
    const r = reports[0];
    const out = await rawFetch("POST", "/client-report", r);
    if (out.kind !== "ok") return;
    reports.shift();
  }
}

}

/**
 * Every Control Center change goes through here. `req.body.request_id` (decisions) is fixed when the
 * owner confirms, so a retry is the same action. Outcome per post: recorded | failed (never ambiguous).
 */
async function mutate(req, { label, key = null, ok = "Done." } = {}) {
  setOutcome(key, { state: "sending", label });
  // 1. Is the Access session accepted right now? (A GET, like the page load.)
  const pre = await rawFetch("GET", "/whoami");
  if (pre.kind === "ok") S.whoami = pre.data;
  if (pre.kind === "signin") {
    evidence("access_redirect", { stage: "preflight", method: "GET", path: "/api/whoami", action: label });
    if (!(await signInKeeping(label))) return failed(key, label, "you cancelled the sign-in");
    await refreshWhoami();
  }
  // 2. The action itself.
  const t0 = Date.now();
  let r = await rawFetch(req.method, req.path, req.body);
  if (r.kind === "signin") {
    evidence("access_redirect", { stage: "action", method: req.method, path: "/api" + req.path, action: label,
      preflight: pre.kind, ms_after_preflight_ok: pre.kind === "ok" ? Date.now() - t0 : null });
    if (!(await signInKeeping(label))) { await flushReports(); return failed(key, label, "you cancelled the sign-in"); }
    await refreshWhoami();
    r = await rawFetch(req.method, req.path, req.body);           // exactly once more, same request id
    evidence(r.kind === "ok" ? "retry_succeeded" : "retry_failed", { method: req.method, path: "/api" + req.path,
      action: label, result: r.kind, status: r.status ?? null });
  }
  await flushReports();
  // LCE-042: another decision is still pending for this post — never replace it without asking.
  if (r.kind === "http" && r.status === 409 && /^pending_conflict/.test(r.error) && req.body && !req.body.replace_pending) {
    const what = (r.error.match(/your (\w+)/) ?? [])[1] ?? "earlier decision";
    const when = (r.error.match(/recorded ([0-9T:+-]+)/) ?? [])[1];
    const yes = await dialog("Replace your pending decision?", h("p", {},
      `Your ${what}${when ? ` (recorded ${localTime(when, tz())})` : ""} is still waiting to be applied. ` +
      `Replace it with this ${label}? If you keep it, nothing else changes.`),
      [{ label: `Replace with ${label}`, cls: "danger", value: () => true }]);
    if (!yes) { setOutcome(key, { state: "kept", label, message: `your pending ${what} stays` }); return null; }
    return mutate({ ...req, body: { ...req.body, replace_pending: true } }, { label, key, ok });
  }
  if (r.kind === "ok") {
    setOutcome(key, { state: "recorded", label, decision_id: r.data?.decision_id ?? null, replayed: Boolean(r.data?.replayed) });
    toast(ok, "ok");
    await load();
    return r.data;
  }
  return failed(key, label, r.kind === "signin" ? SIGNIN : r.kind === "network" ? `the Control Center could not be reached (${r.error})`
    : `${r.error} (HTTP ${r.status})`, () => mutate(req, { label, key, ok }));
}
function failed(key, label, why, retry = null) {
  setOutcome(key, { state: "failed", label, message: why, retry });
  toast(`${cap(label)} failed — the request was not recorded: ${why}`, "err");
  return null;
}
const cap = (x) => x.charAt(0).toUpperCase() + x.slice(1);

// ── dialogs (no window.prompt: works on phones) ─────────────────────
function dialog(title, body, actions) {
  return new Promise((resolve) => {
    const dlg = h("dialog", { class: "dlg" });
    const close = (v) => { dlg.close(); dlg.remove(); resolve(v); };
    dlg.append(h("form", { method: "dialog", onsubmit: (e) => e.preventDefault() },
      h("h3", {}, title), body,
      h("div", { class: "dlg-actions" }, h("button", { type: "button", onclick: () => close(null) }, "Cancel"),
        actions.map((a) => h("button", { type: "button", class: a.cls || "primary", onclick: () => {
          const v = a.value();
          if (v !== undefined) close(v);
        } }, a.label)))));
    document.body.append(dlg);
    dlg.addEventListener("cancel", () => close(null));
    dlg.showModal();
  });
}
function typed(phrase, explain, cls = "primary", label = "Confirm") {
  const input = h("input", { type: "text", autocomplete: "off", autocapitalize: "characters", spellcheck: "false",
    placeholder: phrase, "aria-label": "Confirmation phrase" });
  const err = h("p", { class: "err small" });
  return dialog("Confirm", [h("p", {}, explain), h("p", { class: "small" }, "Type ", h("code", {}, phrase), " to confirm:"), input, err],
    [{ label, cls, value: () => (input.value.trim() === phrase ? phrase : (err.textContent = "The phrase does not match.", undefined)) }]);
}
function askText(title, label, { multiline = false, value = "", required = true, max = 1000 } = {}) {
  const input = multiline ? h("textarea", { rows: 14, maxlength: max }) : h("input", { type: "text", maxlength: max });
  input.value = value;
  const count = h("p", { class: "small muted" });
  const upd = () => { count.textContent = `${input.value.length} / ${max}`; };
  input.addEventListener("input", upd); upd();
  const err = h("p", { class: "err small" });
  return dialog(title, [h("label", { class: "small" }, label), input, count, err],
    [{ label: "Save", value: () => (!required || input.value.trim() ? input.value : (err.textContent = "Required.", undefined)) }]);
}
function askDate(title, label, min) {
  const input = h("input", { type: "date", min });
  const err = h("p", { class: "err small" });
  return dialog(title, [h("label", { class: "small" }, label), input, err],
    [{ label: "Save", value: () => (input.value ? input.value : (err.textContent = "Pick a date.", undefined)) }]);
}
function confirmBox(title, text, label = "Yes", cls = "primary") {
  return dialog(title, h("p", {}, text), [{ label, cls, value: () => true }]);
}

async function act(req, ok = "Done.", label = "change", key = null) {
  return mutate(req, { label, key, ok });
}

const LABELS = { approve: "approval", reject: "rejection", edit: "edit", refresh: "refresh", regenerate: "regeneration request",

function decide(body, ok) {

      ok: ok || "Decision recorded. The private workflow applies it at its next run (within about an hour)." });

  },

function identityBadge() {
  const i = S.identity;
  if (!i) return badge("checking", "muted");
  if (i.ok && i.person_urn_matches === true) return badge("verified", "ok");
  if (i.ok && i.person_urn_matches === null) return badge("verified · profile not set", "warn");
  if (i.ok) return badge("different member", "err");
  return badge(String(i.status || "unverified").replaceAll("_", " "), "err");
}

// LCE-039: a real <img> (eager, with intrinsic size so layout reserves space). If the browser cannot
// load it (e.g. the Access session expired), a visible diagnostic replaces it instead of bare alt text.
function postImage(postId, cls, img, version = null) {
  const src = `/api/posts/${encodeURIComponent(postId)}/` + (version ? `versions/${version}/image` : "image");
  const el = h("img", { class: cls, src, alt: img?.alt || "Post image", width: img?.width || null, height: img?.height || null,
    decoding: "async", loading: "eager", "data-post": postId });
  el.addEventListener("error", async () => {
    let why = "the browser could not load it", signin = false;
    try {
      const r = await fetch(src, { credentials: "same-origin", cache: "no-store", redirect: "manual" });
      if (r.type === "opaqueredirect" || r.status === 0) { signin = true; sessionExpired(); }
      else why = `HTTP ${r.status}, ${r.headers.get("content-type") || "no content type"}`;
    } catch (e) { why = String(e?.message || e); }
    // LCE-047: an Access refusal is the session, not the image: one sign-in (banner) reloads every image.
    el.replaceWith(signin
      ? h("div", { class: "media-note warn" }, "Image waiting for sign-in: Cloudflare Access refused this request because the session ended. It loads again right after you sign in (banner above).")
      : h("div", { class: "media-note warn" }, "The image could not be displayed: ", why, "."));
  }, { once: true });
  return el;
}

// LCE-039: thumbnail inside a list card: the real image, square, uncropped, never wrapped in a link
// (a link around an image is what text renderers print as "[alt](url)"); clicking opens the post.
function cardImage(postId, m) {
  const frame = h("div", { class: "card-media", title: "Open post", onclick: () => { location.hash = `#post/${postId}`; } },
    postImage(postId, "card-img", { alt: m.alt, ...(pipePost(postId)?.image ?? {}) }));
  return frame;
}

function mediaView(p, m, big) {
  if (m.preview === "available") return postImage(p.post_id, big ? "media" : "thumb", { alt: m.alt, ...(pipePost(p.post_id)?.image ?? {}) });
  if (m.type.key === "image") {
    return h("div", { class: "media-note warn" }, "Image recorded", m.file ? ` (${m.file})` : "", " — preview not available yet; it appears after the next private sync.");
  }
  if (!m.type.implemented) return h("div", { class: "media-note warn" }, `${m.type.label}: not implemented — this post cannot be published as is.`);
  return null;
}

const stepText = (v) => {
  if (!v || typeof v !== "object") return String(v ?? "—");
  if ("passed" in v && "review" in v) return `${v.passed} passed · ${v.failed} failed · ${v.review} owner review`;
  if ("compared_against" in v) return `${v.status} · compared with ${v.compared_against} · ${v.exact} exact · ${v.near} near`;
  return Object.entries(v).map(([k, x]) => `${k}: ${x}`).join(" · ");
};

const CLOSED_NOTE = {
  skip: (d) => `Skipped · slot released (recorded ${fmtWhen(d.created_at)}). This version is no longer a candidate and nothing will be regenerated; the repository records it at the next private run and keeps the version in History. `,
  reject: (d) => `Rejected (recorded ${fmtWhen(d.created_at)}). This version is no longer a candidate; the repository records it at the next private run. `,

// What happened to the owner's last action here: sending, recorded, or failed (never ambiguous).
function outcomeLine(key, big = false) {
  const o = key ? S.outcomes[key] : null;
  if (!o) return null;
  const now = outcomeNow(o, S.snap?.decisions);
  if (now?.kind === "resolved") {
    return h("div", { class: "outcome small muted" + (big ? " notice" : "") },
      `Your ${o.label} (recorded ${localTime(o.at, tz())}) was ${now.text}.`);
  }
  if (o.state === "sending") return h("div", { class: "outcome small muted" }, `Sending your ${o.label}…`);
  if (o.state === "kept") return h("div", { class: "outcome small muted" }, `${cap(o.label)} not sent — ${o.message}.`);
  if (o.state === "recorded") {
    return h("div", { class: "outcome ok-text" + (big ? " notice info" : " small") }, `${cap(o.label)} requested — recorded ${localTime(o.at, tz())}`,
      o.replayed ? " (already recorded on the first attempt)" : "", o.decision_id ? h("code", { class: "small" }, " " + o.decision_id.slice(0, 10)) : null);
  }
  return h("div", { class: "outcome notice err" + (big ? "" : " small"), role: "alert" },
    h("strong", {}, `${cap(o.label)} failed — the request was not recorded. `), o.message ?? "",
    o.retry ? [" ", h("button", { type: "button", class: "small", onclick: () => o.retry() }, "Try again")] : null);
}

  document.getElementById("banner").replaceChildren(...(s.emergency_stop ? [h("div", { class: "notice err" }, "Emergency stop is on — nothing publishes. ",
    h("button", { type: "button", class: "small", onclick: () => A.emergency(false) }, "Release…"))] : []));
  const views = { overview: viewOverview, upcoming: viewUpcoming, posts: viewPosts, post: viewPost, test: viewTest, history: viewHistory, system: viewSystem };
  main.replaceChildren(...[(views[view] || viewOverview)(arg)].flat(Infinity).filter(Boolean));
}

async function load() {
  const status = document.getElementById("status");
  if (new URLSearchParams(location.search).get("reauth") === "1") {
    // opened by "Sign in again": Access has signed us in by the time this page runs
    if (window.opener) { window.opener.postMessage("lce-reauth-ok", location.origin); window.close(); }
    main.replaceChildren(card("Signed in", h("p", {}, "You are signed in again. You can close this window.")));
    return;
  }
  refreshWhoami();
  try {
    const [snap, pipe, mig] = await Promise.all([
      api("GET", "/snapshot"),
      api("GET", "/pipeline").catch((e) => (e.status === 404 ? null : Promise.reject(e))),
      api("GET", "/migrations").catch(() => null),
    ]);
    S.snap = snap; S.pipe = pipe; S.migrations = mig; lastLoad = Date.now();
    // LCE-045: this page's code is older than the deployed Control Center: reload instead of mixing
    // old rendering with new data (unless the owner is in the middle of something).
    if (snap.ui_build && !UI_BUILD.startsWith("__") && snap.ui_build !== UI_BUILD) {
      const busy = document.querySelector("dialog[open]") || Object.values(S.outcomes).some((o) => o.state === "sending");
      if (!busy) { location.reload(); return; }
    }
    status.textContent = "Updated " + localTime(snap.now, snap.settings?.timezone);
    status.className = "status ok";
    render();
    if (!S.identity) {
      fetch("/api/linkedin/identity", { credentials: "same-origin", cache: "no-store" })
        .then((r) => r.json()).then((i) => { S.identity = i; render(); })
        .catch(() => { S.identity = { ok: false, status: "unreachable", reason: "identity check failed" }; render(); });
    }
  } catch (err) {
    status.textContent = "No data";
    status.className = "status err";
    main.replaceChildren(card("Could not load", h("p", {}, String(err.message || err)),
      h("p", { class: "small muted" }, "You must be signed in through Cloudflare Access. The Control Center never shows substitute data.")));
  }
}

// Warn as soon as the tab is used again after the Access session ran out (before a dialog is filled in).
let lastPing = 0;
window.addEventListener("focus", () => {
  if (Date.now() - lastPing < 60e3 || document.getElementById("session-banner")) return;
  lastPing = Date.now();
  api("GET", "/whoami").then((w) => { S.whoami = w; }).catch((e) => { if (e.signin) sessionExpired(); });
});

function shell() {
  const nav = (cls, items) => h("nav", { class: cls, "aria-label": "Main" }, items.map(([id, label]) => h("a", { href: "#" + id, "data-nav": id }, label)));
  document.getElementById("sidebar").replaceChildren(nav("side-nav", NAV));
  document.getElementById("bottombar").replaceChildren(nav("bottom-nav", NAV.filter(([id]) => MOBILE_NAV.has(id)).map(([id, l]) => [id, id === "test" ? "Publish" : l])));
  document.getElementById("refresh").addEventListener("click", () => { S.identity = null; load(); });
  document.getElementById("estop").addEventListener("click", () => A.emergency(!st().emergency_stop));
  window.addEventListener("hashchange", () => { if (S.snap) { render(); window.scrollTo({ top: 0 }); } });
}

shell();
load();
// LCE-045: coming back to a tab left open shows the current state, never a stale one.
document.addEventListener("visibilitychange", () => {
  if (document.visibilityState === "visible" && Date.now() - lastLoad > 30000 && !document.querySelector("dialog[open]")) load();
});
