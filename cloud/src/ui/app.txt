// ── tiny DOM helpers ────────────────────────────────────────────────
function h(tag, attrs, ...kids) {
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs || {})) {
    if (v === null || v === undefined || v === false) continue;
    if (k === "class") el.className = v;
    else if (k.startsWith("on")) el.addEventListener(k.slice(2), v);
    else el.setAttribute(k, v === true ? "" : String(v));
  }
  for (const c of kids.flat(Infinity)) {
    if (c === null || c === undefined || c === false) continue;
    el.append(c instanceof Node ? c : document.createTextNode(String(c)));
  }
  return el;
}
const badge = (text, tone) => h("span", { class: "badge " + (tone || "muted") }, text);
const card = (title, ...body) => h("section", { class: "card" }, title ? h("h3", {}, title) : null, ...body);
const empty = (t) => h("p", { class: "empty" }, t);

const fmtWhen = (iso) => iso ? `${localDate(iso, tz())} ${localTime(iso, tz())}` : UNKNOWN;
const link = (href, text, cls) => h("a", { href, class: cls }, text);
const ext = (href, text) => h("a", { href, target: "_blank", rel: "noopener noreferrer" }, text);
const short = (x, n = 12) => (x ? String(x).slice(0, n) : "—");

function toast(text, tone) {
  document.getElementById("toast")?.remove();
  const t = h("div", { id: "toast", role: "status", class: tone || "" }, text);
  document.body.append(t);
  setTimeout(() => t.remove(), 7000);
}

async function api(method, path, body) {
  const res = await fetch("/api" + path, {
    method, credentials: "same-origin", cache: "no-store",
    headers: { "content-type": "application/json", "x-lce-client": "dashboard" },
    body: body === undefined ? undefined : JSON.stringify(body),
  });
  let data = {};
  try { data = await res.json(); } catch { data = {}; }
  if (!res.ok) {
    const err = new Error(data.error || data.detail || data.reason || ("HTTP " + res.status));
    err.status = res.status;
    err.data = data;
    throw err;
  }
  return data;
}

// ── dialogs (no window.prompt: works on phones) ─────────────────────
function dialog(title, body, actions) {
  return new Promise((resolve) => {
    const dlg = h("dialog", { class: "dlg" });
    const close = (v) => { dlg.close(); dlg.remove(); resolve(v); };
    dlg.append(h("form", { method: "dialog", onsubmit: (e) => e.preventDefault() },
      h("h3", {}, title), body,
      h("div", { class: "dlg-actions" }, h("button", { type: "button", onclick: () => close(null) }, "Cancel"),
        actions.map((a) => h("button", { type: "button", class: a.cls || "primary", onclick: () => {
          const v = a.value();
          if (v !== undefined) close(v);
        } }, a.label)))));
    document.body.append(dlg);
    dlg.addEventListener("cancel", () => close(null));
    dlg.showModal();
  });
}
function typed(phrase, explain, cls = "primary", label = "Confirm") {
  const input = h("input", { type: "text", autocomplete: "off", autocapitalize: "characters", spellcheck: "false",
    placeholder: phrase, "aria-label": "Confirmation phrase" });
  const err = h("p", { class: "err small" });
  return dialog("Confirm", [h("p", {}, explain), h("p", { class: "small" }, "Type ", h("code", {}, phrase), " to confirm:"), input, err],
    [{ label, cls, value: () => (input.value.trim() === phrase ? phrase : (err.textContent = "The phrase does not match.", undefined)) }]);
}
function askText(title, label, { multiline = false, value = "", required = true, max = 1000 } = {}) {
  const input = multiline ? h("textarea", { rows: 14, maxlength: max }) : h("input", { type: "text", maxlength: max });
  input.value = value;
  const count = h("p", { class: "small muted" });
  const upd = () => { count.textContent = `${input.value.length} / ${max}`; };
  input.addEventListener("input", upd); upd();
  const err = h("p", { class: "err small" });
  return dialog(title, [h("label", { class: "small" }, label), input, count, err],
    [{ label: "Save", value: () => (!required || input.value.trim() ? input.value : (err.textContent = "Required.", undefined)) }]);
}
function askDate(title, label, min) {
  const input = h("input", { type: "date", min });
  const err = h("p", { class: "err small" });
  return dialog(title, [h("label", { class: "small" }, label), input, err],
    [{ label: "Save", value: () => (input.value ? input.value : (err.textContent = "Pick a date.", undefined)) }]);
}
function confirmBox(title, text, label = "Yes", cls = "primary") {
  return dialog(title, h("p", {}, text), [{ label, cls, value: () => true }]);
}

async function act(fn, ok = "Done.") {
  try {
    const out = await fn();
    if (out === null || out === undefined) return;
    toast(ok, "ok");
    await load();
  } catch (err) {
    toast("Refused: " + err.message, "err");
  }
}

function decide(body, ok) {
  return act(() => api("POST", "/decisions", body), ok ||
    "Decision recorded. The private workflow applies it at its next run (within about an hour).");
}

// ── actions ─────────────────────────────────────────────────────────
const A = {
  async approve(p) {
    const phrase = await typed(`APPROVE ${p.post_id}`, `Approve this exact text (hash ${short(p.actual_hash)}…). ` +
      "Approval does not publish: the post then waits in the cloud queue until you schedule or test-publish it.");
    if (phrase) decide({ action: "approve", post_id: p.post_id, content_hash: p.actual_hash, confirm: phrase });
  },
  async reject(p) {
    const reason = await askText("Reject", "Why? (kept in the post history)", { max: 500 });
    if (reason === null) return;
    const phrase = await typed(`REJECT ${p.post_id}`, "The post will not be published.", "danger", "Reject");
    if (phrase) decide({ action: "reject", post_id: p.post_id, reason, confirm: phrase });
  },
  async edit(p) {
    const text = await askText("Edit post", "Post text (QA and the duplicate check run again; a new approval is needed)",
      { multiline: true, value: p.text ?? "", max: 3000 });
    if (text !== null) decide({ action: "edit", post_id: p.post_id, base_hash: p.actual_hash, text });
  },
  async regenerate(p) {
    const reason = await askText("Request regeneration", "What should change? The next drafting session rewrites it; nothing is regenerated automatically.");
    if (reason !== null) decide({ action: "regenerate", post_id: p.post_id, reason });
  },
  async reschedule(p) {
    const date = await askDate("Reschedule", "New planned date", localDate(S.snap.now, tz()));
    if (date) decide({ action: "reschedule", post_id: p.post_id, date });
  },
  async skip(p, planDate) {
    const reason = await askText("Skip", "Reason (optional)", { required: false, max: 500 });
    if (reason === null) return;
    decide(p ? { action: "skip", post_id: p.post_id, reason } : { action: "skip", plan_date: planDate, reason });
  },
  async duplicate(p) {
    const date = await askDate("Duplicate", "Planned date for the copy", localDate(S.snap.now, tz()));
    if (date) decide({ action: "duplicate", post_id: p.post_id, date });
  },
  cancel: (d) => act(() => api("DELETE", "/decisions/" + encodeURIComponent(d.decision_id)), "Decision cancelled."),
  async schedule(postId, slot) {
    const phrase = await typed(`SCHEDULE ${postId}`, `Publish ${postId} automatically at ${slot.local} — only while auto-publish is ON and the emergency stop is off.`);
    if (phrase) act(() => api("POST", "/consents", { post_id: postId, slot_id: slot.slot_id, confirm: phrase }), "Scheduled.");
  },
  async revoke(c) {
    if (await confirmBox("Revoke", "Remove this post from its slot? It stays approved in the cloud queue.", "Revoke", "danger")) {
      act(() => api("DELETE", "/consents/" + encodeURIComponent(c.consent_id)), "Consent revoked.");
    }
  },
  async postAction(postId, action, explain, extra) {
    const phrase = await typed(`${action.toUpperCase()} ${postId}`, explain);
    if (phrase) act(() => api("POST", `/posts/${encodeURIComponent(postId)}/${action}`, { confirm: phrase, ...(extra || {}) }));
  },
  async reconcile(postId) {
    const url = await askText("Reconcile", "If the post IS on LinkedIn, paste its URL. Leave empty if it is NOT.", { required: false, max: 500 });
    if (url === null) return;
    const extra = url.trim() ? { decision: "published", url: url.trim() } : { decision: "not_published" };
    A.postAction(postId, "reconcile", url.trim() ? `Record ${postId} as published at ${url.trim()}.` : `Record that ${postId} is NOT on LinkedIn.`, extra);
  },
  async autoPublish(on) {
    if (!on) {
      if (await confirmBox("Auto-publish OFF", "Scheduled posts will not be published until it is on again.", "Turn off", "danger")) {
        act(() => api("PUT", "/settings", { auto_publish: false }), "Auto-publish is OFF.");
      }
      return;
    }
    const phrase = await typed("ENABLE AUTO-PUBLISH", "Scheduled, consented posts will then be published automatically at their slots.");
    if (phrase) act(() => api("PUT", "/settings", { auto_publish: true, confirm: phrase }), "Auto-publish is ON.");
  },
  async emergency(on) {
    if (on) {
      if (await confirmBox("Emergency stop", "Block every publication now (scheduled and manual)?", "Stop everything", "danger")) {
        act(() => api("PUT", "/settings", { emergency_stop: true }), "Emergency stop is ON: nothing publishes.");
      }
      return;
    }
    const phrase = await typed("RELEASE EMERGENCY STOP", "Publications become possible again (still only with approval, consent and auto-publish, or a test publish).");
    if (phrase) act(() => api("PUT", "/settings", { emergency_stop: false, confirm: phrase }), "Emergency stop released.");
  },
};

function identityBadge() {
  const i = S.identity;
  if (!i) return badge("checking…", "muted");
  if (i.ok && i.person_urn_matches === true) return badge("verified", "ok");
  if (i.ok && i.person_urn_matches === null) return badge("verified · URN not set", "warn");
  if (i.ok) return badge("member ≠ configured URN", "err");
  return badge(String(i.status || "unverified").replaceAll("_", " "), "err");
}

function feedPreview(p, cp) {
  const s = st();
  const text = cp?.text ?? p?.text ?? "";
  const type = contentType({ ...(cp ?? {}), ...(p ?? {}), image: p?.image ?? cp?.image });
  let media = null;
  if (cp?.image) media = h("img", { class: "media", src: `/api/posts/${encodeURIComponent(cp.post_id)}/image`, alt: p?.image?.alt_text || "post image" });
  else if (type.key === "image") {
    media = h("div", { class: "media placeholder" }, "▣ ", p?.image?.file || "image",
      h("div", { class: "small" }, p?.image?.alt_text ? "Alt: " + p.image.alt_text : "No alt text recorded"),

  } else if (!type.implemented) media = h("div", { class: "media placeholder unsupported" }, type.icon, " ", type.label, " — not implemented yet");
  return h("article", { class: "feed" },
    h("header", { class: "feed-head" }, h("div", { class: "avatar" }, initials(s.display_name)),
      h("div", {}, h("div", { class: "feed-name" }, targetName()),
        h("div", { class: "small muted" }, "Personal profile · ", (s.visibility || "PUBLIC").toLowerCase() === "public" ? "🌐 Anyone" : s.visibility))),
    text ? h("div", { class: "feed-text" }, segments(text).map((x) => x.tag ? h("span", { class: "tag" }, x.text) : x.text))

    media,
    h("footer", { class: "feed-foot small muted" }, typeChip(type), " · ", `${text.trim().length} / 3000 characters`,
      hashtags(text).length ? ` · ${hashtags(text).length} hashtag(s)` : ""));
}

function weekStrip(up) {
  return h("div", { class: "week" }, up.days.map((d) => h("div", { class: "day" + (d.date === up.from ? " today" : "") },
    h("div", { class: "day-head" }, weekday(d.date), h("span", { class: "muted" }, " " + d.date.slice(5))),
    d.items.filter((i) => i.kind !== "slot").map((i) => h("a", { class: "chip " + i.kind, href: i.post_id ? `#post/${i.post_id}` : "#upcoming" },
      i.time ? h("b", {}, i.time + " ") : null, i.type ? i.type.icon + " " : "", i.title)),
    d.items.some((i) => i.kind !== "slot") ? null : h("div", { class: "small muted" }, "—"))));
}

function agendaItem(i) {
  if (i.kind === "slot") {
    const ready = (S.snap.posts ?? []).filter((p) => p.state === "READY_TO_PUBLISH" && !activeConsent(p.post_id));
    const slot = (S.snap.upcoming_slots ?? []).find((x) => x.slot_id === i.slot_id);
    let control = h("span", { class: "small muted" }, "no approved post in the cloud queue");
    if (ready.length && slot) {
      const sel = h("select", { "aria-label": "Post for this slot" }, ready.map((p) => h("option", { value: p.post_id }, hook(p.text, 50) ?? p.post_id)));
      control = h("span", { class: "inline" }, sel, h("button", { type: "button", onclick: () => A.schedule(sel.value, slot) }, "Schedule…"));
    }
    return h("div", { class: "agenda-item slot" }, h("div", { class: "agenda-time" }, i.time), h("div", { class: "agenda-body" }, h("div", { class: "muted" }, "Free slot"), control));
  }
  const p = i.post_id ? pipePost(i.post_id) : null;
  return h("div", { class: "agenda-item " + i.kind },
    h("div", { class: "agenda-time" }, i.time ?? "—"),
    h("div", { class: "agenda-body" },
      h("div", { class: "agenda-title" }, i.post_id ? link(`#post/${i.post_id}`, i.title) : i.title),
      h("div", { class: "meta" }, typeChip(i.type), badge(i.kind, i.kind === "scheduled" ? "info" : "muted"),
        i.state ? stateBadge(i.state) : i.plan_status && i.plan_status !== i.kind ? badge(i.plan_status, "muted") : null,
        badge("approval: " + i.approval, i.approval === "approved" ? "ok" : i.approval === "pending" ? "info" : "muted"),
        badge(i.publishing, i.publishing === "published" ? "ok" : "muted")),
      h("div", { class: "actions" },
        i.post_id ? link(`#post/${i.post_id}`, "Preview", "btn") : null,
        p?.state === "AWAITING_APPROVAL" ? h("button", { type: "button", class: "primary", onclick: () => A.approve(p) }, "Approve…") : null,
        i.kind === "planned" && !i.post_id ? h("button", { type: "button", onclick: () => A.skip(null, i.date) }, "Skip…") : null,
        i.kind === "scheduled" ? h("button", { type: "button", onclick: () => A.revoke({ consent_id: i.consent_id }) }, "Unschedule…") : null)));
}

function viewPost(id) {
  const p = pipePost(id), c = cloudPost(id);
  if (!p && !c) return [card("Not found", h("p", {}, `No post ${id} in the mirror or the cloud queue.`), link("#posts", "Back to posts"))];
  const d = pendingFor(id);
  const consent = activeConsent(id);
  const pb = pub(id);
  const inCloud = c && c.state !== "WITHDRAWN";
  const state = p?.state;

function viewTest(id) {
  const s = st();
  const ready = (S.snap.posts ?? []).filter((p) => p.state === "READY_TO_PUBLISH" && !activeConsent(p.post_id));
  const c = id ? cloudPost(id) : null;
  const checks = [
    ["Provider: LinkedIn API", s.provider === "linkedin_api"],
    ["Person URN configured", Boolean(s.person_urn)],
    ["Token present (value never shown)", Boolean(s.token_present)],
    ["Token belongs to the configured personal profile", S.identity?.ok === true && S.identity.person_urn_matches === true],
    ["Emergency stop off", !s.emergency_stop],
  ];
  const allOk = checks.every(([, ok]) => ok);
  const pick = h("div", { class: "cards" }, ready.length ? ready.map((p) => h("a", { class: "post-card" + (p.post_id === id ? " selected" : ""), href: `#test/${p.post_id}` },
    h("div", { class: "post-card-title" }, hook(p.text) ?? p.post_id), h("div", { class: "meta" }, typeChip(contentType({ ...p, ...pipePost(p.post_id) })), h("code", {}, short(p.approved_hash)))))
    : empty("No approved post is waiting in the cloud queue (approve one first; scheduled posts must be unscheduled)."));
  const parts = [
    h("h2", {}, "Controlled test publish"),
    h("div", { class: "notice warn" }, "This sends ONE post to LinkedIn now, as your personal profile. It never runs automatically, never changes auto-publish, and needs your typed confirmation twice."),
    card("1 · Pre-flight", h("ul", { class: "checks" }, checks.map(([label, ok]) => h("li", {}, badge(ok ? "ok" : "blocked", ok ? "ok" : "err"), " ", label)))),
    card("2 · Choose an approved post", pick),
  ];
  if (c && c.state === "READY_TO_PUBLISH") {
    const type = contentType({ ...c, ...pipePost(c.post_id) });
    const input = h("input", { type: "text", autocomplete: "off", spellcheck: "false", placeholder: `PUBLISH NOW ${c.post_id}`, "aria-label": "Confirmation phrase" });
    const go = h("button", { type: "button", class: "danger solid", disabled: true }, "Publish to LinkedIn now…");
    input.addEventListener("input", () => { go.disabled = !(allOk && type.implemented && input.value.trim() === `PUBLISH NOW ${c.post_id}`); });
    go.addEventListener("click", async () => {
      const yes = await confirmBox("Last check before POST", `Send this exact post (hash ${short(c.approved_hash)}…) to LinkedIn now, as ${targetName()} (personal profile)? This cannot be undone from here.`, "Send now", "danger");
      if (!yes) return;
      go.disabled = true;
      try {
        const r = await api("POST", `/posts/${encodeURIComponent(c.post_id)}/publish-now`, { approved_hash: c.approved_hash, confirm: input.value.trim() });
        toast(r.outcome === "published" ? "Published on LinkedIn." : `Result: ${r.outcome}`, r.outcome === "published" ? "ok" : "warn");
      } catch (err) {
        toast("Not published: " + err.message, "err");
      }
      await load();
    });
    parts.push(
      card("3 · Exact final post", publishingAs(), feedPreview(pipePost(c.post_id), c),
        h("p", { class: "small muted" }, "Media: ", type.key === "image" ? `one image (${Math.round((c.image?.bytes ?? 0) / 1024)} KB)` : "none (text-only)",
          type.implemented ? "" : " — this content type is not implemented and cannot be published")),
      card("4 · Confirm", h("p", { class: "small" }, "Type ", h("code", {}, `PUBLISH NOW ${c.post_id}`), " then press the button; a final dialog appears before anything is sent."),
        input, go, allOk ? null : h("p", { class: "err small" }, "Blocked: fix the pre-flight items first.")));
  }
  const pb = id ? pub(id) : null;
  if (pb) {
    parts.push(card("Result", h("dl", { class: "kv" }, h("dt", {}, "State"), h("dd", {}, stateBadge(pb.state)),
      h("dt", {}, "LinkedIn post ID"), h("dd", {}, h("code", {}, pb.remote_id || "—")),
      h("dt", {}, "URL"), h("dd", {}, pb.url ? ext(pb.url, pb.url) : "—"),
      h("dt", {}, "Idempotency key"), h("dd", {}, h("code", {}, short(pb.idempotency_key, 16))),
      h("dt", {}, "Published at"), h("dd", {}, pb.published_at ? fmtWhen(pb.published_at) : "—"))));
  }
  return parts;
}

const NAV = [["overview", "Overview", "◉"], ["upcoming", "Upcoming", "▦"], ["posts", "Posts", "▤"], ["test", "Test publish", "➤"],
  ["history", "History", "↺"], ["system", "System", "⚙"]];

function render() {
  const { view, arg } = parseHash(location.hash);
  for (const a of document.querySelectorAll("[data-nav]")) a.classList.toggle("on", a.dataset.nav === (view === "post" ? "posts" : view));
  const s = st();
  const estop = document.getElementById("estop");
  estop.textContent = s.emergency_stop ? "Release stop…" : "Emergency stop";
  estop.classList.toggle("solid", !s.emergency_stop);
  const banner = document.getElementById("banner");
  banner.replaceChildren(...(s.emergency_stop ? [h("div", { class: "notice err" }, "EMERGENCY STOP ON — nothing publishes. ",
    h("button", { type: "button", onclick: () => A.emergency(false) }, "Release…"))] : []));
  const views = { overview: viewOverview, upcoming: viewUpcoming, posts: viewPosts, post: viewPost, test: viewTest,
    history: viewHistory, system: viewSystem };
  main.replaceChildren(...[(views[view] || viewOverview)(arg)].flat(Infinity).filter(Boolean));
  window.scrollTo({ top: 0 });
}

async function load() {
  const status = document.getElementById("status");
  try {
    const [snap, pipe, mig] = await Promise.all([
      api("GET", "/snapshot"),
      api("GET", "/pipeline").catch((e) => (e.status === 404 ? null : Promise.reject(e))),
      api("GET", "/migrations").catch(() => null),
    ]);
    S.snap = snap; S.pipe = pipe; S.migrations = mig; S.loadedAt = new Date().toISOString();
    status.textContent = "Updated " + localTime(snap.now, snap.settings?.timezone);
    status.className = "badge ok";
    render();
    if (!S.identity) {
      fetch("/api/linkedin/identity", { credentials: "same-origin", cache: "no-store" })
        .then((r) => r.json()).then((i) => { S.identity = i; render(); })
        .catch(() => { S.identity = { ok: false, status: "unreachable", reason: "identity check failed" }; render(); });
    }
  } catch (err) {
    status.textContent = "No data";
    status.className = "badge err";
    main.replaceChildren(card("Could not load", h("p", {}, String(err.message || err)),
      h("p", { class: "small muted" }, "You must be signed in through Cloudflare Access. The Control Center never shows substitute data.")));
  }
}

function shell() {
  const nav = (cls) => h("nav", { class: cls, "aria-label": "Main" }, NAV.map(([id, label, icon]) =>
    h("a", { href: "#" + id, "data-nav": id }, h("span", { class: "nav-icon", "aria-hidden": "true" }, icon), h("span", { class: "nav-label" }, label))));
  document.getElementById("sidebar").replaceChildren(nav("side-nav"));
  document.getElementById("bottombar").replaceChildren(nav("bottom-nav"));
  document.getElementById("refresh").addEventListener("click", () => { S.identity = null; load(); });
  document.getElementById("estop").addEventListener("click", () => A.emergency(!st().emergency_stop));
  window.addEventListener("hashchange", () => { if (S.snap) render(); });
}

shell();
